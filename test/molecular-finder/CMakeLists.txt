cmake_minimum_required(VERSION 3.20)
project(MolecularFinderRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Geant4 11.4.3 EXACT REQUIRED)
include(${Geant4_USE_FILE})
add_executable(FinderOracle FinderOracle.cc)
target_link_libraries(FinderOracle PRIVATE ${Geant4_LIBRARIES})
add_executable(JobLauncher JobLauncher.cc)
target_link_libraries(JobLauncher PRIVATE psapi)
get_filename_component(root "../.." ABSOLUTE)
file(GLOB_RECURSE sources "${root}/src/*.cc")
list(FILTER sources EXCLUDE REGEX "/(MesoSpatialFile|Hdf5TableFile)[.]cc$")
add_executable(NativeContacts NativeContacts.cc ${sources})
target_include_directories(NativeContacts PRIVATE "${root}/header")
target_compile_definitions(NativeContacts PRIVATE DNACHEM_HAS_HDF5=0)
target_link_libraries(NativeContacts PRIVATE ${Geant4_LIBRARIES})

add_executable(OwnershipTrace OwnershipTrace.cc)
target_link_libraries(OwnershipTrace PRIVATE ${Geant4_LIBRARIES})
# Diagnostic-only copy; the shared source and installed headers are never edited.
if(FINDER_BASELINE_SOURCE)
  set(headers "${FINDER_BASELINE_SOURCE}/source/processes/electromagnetic/dna/management/include")
  file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/trace-headers")
  foreach(name G4Octree.hh G4Octree.icc G4OctreeFinder.hh G4OctreeFinder.icc)
    configure_file("${headers}/${name}" "${CMAKE_BINARY_DIR}/trace-headers/${name}" COPYONLY)
  endforeach()
  file(READ "${headers}/G4Octree.icc" trace)
  set(boundary "    if (current_depth > max_depth)")
  string(FIND "${trace}" "${boundary}" found)
  if(found EQUAL -1)
    message(FATAL_ERROR "Trace boundary absent")
  endif()
  string(REPLACE "${boundary}" "    static size_t diagnosticNodes = 0;\n    ++diagnosticNodes;\n    if (diagnosticNodes <= 12 || diagnosticNodes % 100 == 0)\n        std::cerr << \"TRACE node \" << diagnosticNodes << \" depth \" << current_depth << \" entries \" << input_values.size() << \" box \" << box << std::endl;\n    if (diagnosticNodes == 1000) throw std::runtime_error(\"DIAGNOSTIC_NODE_CAP\");\n${boundary}" trace "${trace}")
  file(WRITE "${CMAKE_BINARY_DIR}/trace-headers/G4Octree.icc" "${trace}")
  add_executable(FinderTrace FinderOracle.cc)
  target_include_directories(FinderTrace BEFORE PRIVATE "${CMAKE_BINARY_DIR}/trace-headers")
  target_link_libraries(FinderTrace PRIVATE ${Geant4_LIBRARIES})
endif()
