cmake_minimum_required(VERSION 3.20)
project(TrackPackets LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Geant4 11.4.3 EXACT REQUIRED)
include(${Geant4_USE_FILE})
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(READ "${root}/src/chemistry/TimeStepAction.cc" chemistry)
string(REPLACE "void TimeStepAction::StartProcessing()\n{" "void TimeStepAction::StartProcessing()\n{\n CapturePrecursors();" chemistry "${chemistry}")
file(WRITE "${CMAKE_BINARY_DIR}/TimeStepAction.cc" "void CapturePrecursors();\n${chemistry}")
file(GLOB_RECURSE sources "${root}/src/*.cc")
list(FILTER sources EXCLUDE REGEX "/(TimeStepAction|MesoSpatialFile|Hdf5TableFile)[.]cc$")
add_executable(TrackCapture TrackCapture.cc ${sources} "${CMAKE_BINARY_DIR}/TimeStepAction.cc")
target_include_directories(TrackCapture PRIVATE "${root}/header")
target_compile_definitions(TrackCapture PRIVATE DNACHEM_HAS_HDF5=0)
target_link_libraries(TrackCapture PRIVATE ${Geant4_LIBRARIES})
enable_testing()
add_test(NAME PrecursorBoundary COMMAND TrackCapture "${CMAKE_CURRENT_SOURCE_DIR}/Capture.in")
set_tests_properties(PrecursorBoundary PROPERTIES LABELS injection-packets TIMEOUT 120)
