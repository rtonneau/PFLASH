cmake_minimum_required(VERSION 3.20)
project(TrackPackets LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Geant4 11.4.3 EXACT REQUIRED)
include(${Geant4_USE_FILE})
get_filename_component(root "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
file(GLOB_RECURSE sources "${root}/src/*.cc")
set(fingerprint "Geant4-11.4.3;PureWater;IRT_syn;water-500um;500keV;parallel-uniform-face;no-energy-cuts;startup-1ps;internal-mm-ns-MeV")
file(GLOB_RECURSE headers "${root}/header/*.hh")
foreach(source IN LISTS sources headers)
  file(SHA256 "${source}" hash)
  string(APPEND fingerprint "${hash}")
endforeach()
string(SHA256 fingerprint "${fingerprint}")
function(instrument file signature hook declaration)
  file(READ "${root}/src/${file}.cc" content)
  string(FIND "${content}" "${signature}\n{" found)
  if(found EQUAL -1)
    message(FATAL_ERROR "Instrumentation boundary missing: ${file}")
  endif()
  string(REPLACE "${signature}\n{" "${signature}\n{\n  ${hook}" content "${content}")
  get_filename_component(name "${file}" NAME)
  file(WRITE "${CMAKE_BINARY_DIR}/${name}.cc" "${declaration}\n${content}")
endfunction()
instrument("chemistry/TimeStepAction" "void TimeStepAction::StartProcessing()" "CapturePrecursors();" "void CapturePrecursors();\nvoid ProbeEndChemistry();")
file(READ "${CMAKE_BINARY_DIR}/TimeStepAction.cc" content)
string(REPLACE "void TimeStepAction::EndProcessing()\n{" "void TimeStepAction::EndProcessing()\n{\n ProbeEndChemistry();" content "${content}")
file(WRITE "${CMAKE_BINARY_DIR}/TimeStepAction.cc" "${content}")
instrument("actions/PrimaryGeneratorAction" "void PrimaryGeneratorAction::GeneratePrimaries(G4Event *anEvent)" "ProbePrimary(fpParticleGun);" "class G4ParticleGun; void ProbePrimary(G4ParticleGun*);")
instrument("actions/SteppingAction" "void SteppingAction::UserSteppingAction(const G4Step *step)" "ProbeStep(step);" "class G4Step; void ProbeStep(const G4Step*);")
instrument("actions/TrackingAction" "void TrackingAction::PostUserTrackingAction(const G4Track *track)" "ProbeTrack(track);" "class G4Track; void ProbeTrack(const G4Track*);")
file(READ "${root}/src/actions/EventAction.cc" content)
string(REGEX MATCH "void EventAction::EndOfEventAction[^\n]*" signature "${content}")
instrument("actions/EventAction" "${signature}" "ProbeEvent(event);" "class G4Event; void ProbeEvent(const G4Event*);")
file(READ "${root}/src/scoring/PrimaryKiller.cc" killer)
string(REPLACE "((G4Track *)track)->SetTrackStatus(G4TrackStatus::fStopAndKill);" "ProbeTruncated(); ((G4Track *)track)->SetTrackStatus(G4TrackStatus::fStopAndKill);" killer "${killer}")
file(WRITE "${CMAKE_BINARY_DIR}/PrimaryKiller.cc" "void ProbeTruncated();\n${killer}")
list(FILTER sources EXCLUDE REGEX "/(TrackingAction|PrimaryKiller|TimeStepAction|PrimaryGeneratorAction|SteppingAction|EventAction|MesoSpatialFile|Hdf5TableFile)[.]cc$")
add_library(PacketData TrackPacket.cc)
add_executable(TrackCapture TrackCapture.cc ${sources} "${CMAKE_BINARY_DIR}/TrackingAction.cc" "${CMAKE_BINARY_DIR}/PrimaryKiller.cc" "${CMAKE_BINARY_DIR}/TimeStepAction.cc" "${CMAKE_BINARY_DIR}/PrimaryGeneratorAction.cc" "${CMAKE_BINARY_DIR}/SteppingAction.cc" "${CMAKE_BINARY_DIR}/EventAction.cc")
target_include_directories(TrackCapture PRIVATE "${root}/header")
target_compile_definitions(TrackCapture PRIVATE DNACHEM_HAS_HDF5=0 PACKET_COMPATIBILITY="${fingerprint}")
target_link_libraries(TrackCapture PRIVATE PacketData ${Geant4_LIBRARIES})
enable_testing()

target_link_libraries(TrackPacketTest PRIVATE PacketData)

set_tests_properties(PacketAndSource PROPERTIES LABELS injection-packets)
add_executable(PacketComparison PacketComparison.cc BeamSchedule.cc)
target_link_libraries(PacketComparison PRIVATE PacketData)
add_test(NAME LibraryHoldoutChemistry COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:TrackCapture> -DCOMPARE=$<TARGET_FILE:PacketComparison> -DMACRO=${CMAKE_CURRENT_SOURCE_DIR}/Capture.in -DOUT=${CMAKE_BINARY_DIR}/validation -P ${CMAKE_CURRENT_SOURCE_DIR}/RunPacketTests.cmake)
set_tests_properties(LibraryHoldoutChemistry PROPERTIES LABELS injection-packets TIMEOUT 900)
add_executable(PacketInvalid PacketInvalid.cc)
target_link_libraries(PacketInvalid PRIVATE PacketData)
add_test(NAME PacketRejections COMMAND ${CMAKE_COMMAND} -DEXE=$<TARGET_FILE:TrackCapture> -DMUTATE=$<TARGET_FILE:PacketInvalid> -DMACRO=${CMAKE_CURRENT_SOURCE_DIR}/Capture.in -DOUT=${CMAKE_BINARY_DIR}/validation -P ${CMAKE_CURRENT_SOURCE_DIR}/RunRejectionTests.cmake)
set_tests_properties(PacketRejections PROPERTIES LABELS injection-packets DEPENDS LibraryHoldoutChemistry TIMEOUT 180)
