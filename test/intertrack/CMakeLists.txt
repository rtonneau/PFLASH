cmake_minimum_required(VERSION 3.20)
project(Intertrack LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
set(probe ../timed-injection/probe)
add_library(Intertrack Intertrack.cc ${probe}/CalibrationStatistics.cc ${probe}/CaptureEvidence.cc ${probe}/TrackPacket.cc ${probe}/BeamSchedule.cc)
target_include_directories(Intertrack PUBLIC . ${probe})
add_executable(IntertrackTest IntertrackTest.cc)
target_link_libraries(IntertrackTest PRIVATE Intertrack)
add_test(NAME IntertrackTest COMMAND IntertrackTest)
add_executable(ArrivalAudit ArrivalAudit.cc)
add_executable(MatchedStudy MatchedStudy.cc)
target_link_libraries(MatchedStudy PRIVATE Intertrack)
add_executable(EvidenceTest EvidenceTest.cc)
target_link_libraries(EvidenceTest PRIVATE Intertrack)
add_test(NAME EvidenceTest COMMAND EvidenceTest ${CMAKE_BINARY_DIR}/evidence-tests)
find_program(POWERSHELL NAMES pwsh powershell REQUIRED)
add_test(NAME BoundedProcessTest COMMAND ${POWERSHELL} -NoProfile -File ${CMAKE_CURRENT_SOURCE_DIR}/TestBoundedProcess.ps1 -Root ${CMAKE_BINARY_DIR}/runner-tests)
